use three_d::*;
pub struct OrbitCamera {
    pub camera: Camera,
    control: OrbitControl,
    target: Vec3,
    fov: Degrees,
}

impl OrbitCamera {
    pub fn new(viewport: Viewport, centre: Vec3, radius: f32) -> Self {
        let fov = degrees(45.0);

        let camera = Camera::new_perspective(
            viewport,
            centre + vec3(radius * 2.5, radius * 1.5, radius * 2.5),
            centre,
            vec3(0.0, 1.0, 0.0),
            fov,
            0.01,
            radius * 20.0,
        );

        let control = OrbitControl::new(centre, radius * 0.1, radius * 20.0);

        Self { camera, control, target: centre, fov }
    }

    pub fn update(&mut self, events: &mut [Event]) -> bool {
        self.control.handle_events(&mut self.camera, events)
    }

    pub fn resize(&mut self, viewport: Viewport) {
        self.camera.set_viewport(viewport);
    }

    pub fn recentre(&mut self, centre: Vec3, radius: f32) {
        self.camera.set_view(
            centre + vec3(radius * 2.5, radius * 1.5, radius * 2.5),
            centre,
            vec3(0.0, 1.0, 0.0),
        );
        self.control = OrbitControl::new(centre, radius * 0.1, radius * 20.0);
        self.target = centre;
    }

    /// Call once per frame after `update()` to keep near/far tight
    /// around the current orbit distance — fixes z-fighting that gets
    /// worse the farther the near plane is from the far plane.
    pub fn tighten_clip_planes(&mut self, radius: f32) {
        let distance = self.target.distance(self.camera.position());
        let near = (distance - radius * 2.0).max(radius * 0.001);
        let far = distance + radius * 4.0;
        self.camera.set_perspective_projection(self.fov, near, far);
    }
}```

use three_d::*;

pub struct DebugRenderer {
    dirty: bool,

    point_depth: PointCloud,
    point_overlay: PointCloud,

    line_depth: LineCloud,
    line_overlay: LineCloud,
}

impl DebugRenderer {
    pub fn new() -> Self {
        Self {
            dirty: false,

            point_depth: PointCloud::new(),
            point_overlay: PointCloud::new(),

            line_depth: LineCloud::new(),
            line_overlay: LineCloud::new(),
        }
    }

    pub fn clear(&mut self) {
        self.point_depth.clear();
        self.point_overlay.clear();

        self.line_depth.clear();
        self.line_overlay.clear();

        self.dirty = true;
    }

    pub fn point(
        &mut self,
        position: Vec3,
        radius: f32,
        color: Srgba,
        depth_test: bool,
    ) {
        self.dirty = true;

        let cloud = if depth_test {
            &mut self.point_depth
        } else {
            &mut self.point_overlay
        };

        cloud.push(position, radius, color);
    }

    pub fn edge(
        &mut self,
        a: Vec3,
        b: Vec3,
        thickness: f32,
        color: Srgba,
        depth_test: bool,
    ) {
        self.dirty = true;

        let cloud = if depth_test {
            &mut self.line_depth
        } else {
            &mut self.line_overlay
        };

        cloud.push(a, b, thickness, color);
    }

    pub fn upload(&mut self, context: &Context) {
        if !self.dirty {
            return;
        }

        self.point_depth.upload(context);
        self.point_overlay.upload(context);

        self.line_depth.upload(context);
        self.line_overlay.upload(context);

        self.dirty = false;
    }

    pub fn occluded<'a>(
        &'a self,
    ) -> impl Iterator<Item = &'a dyn Object> {

        self.point_depth
            .object
            .iter()
            .map(|o| o as &dyn Object)
            .chain(
                self.line_depth
                    .object
                    .iter()
                    .map(|o| o as &dyn Object)
            )
    }

    pub fn overlay<'a>(
        &'a self,
    ) -> impl Iterator<Item = &'a dyn Object> {

        self.point_overlay
            .object
            .iter()
            .map(|o| o as &dyn Object)
            .chain(
                self.line_overlay
                    .object
                    .iter()
                    .map(|o| o as &dyn Object)
            )
    }
}

struct PointInstance {
    position: Vec3,
    radius: f32,
    color: Srgba,
}

struct LineInstance {
    a: Vec3,
    b: Vec3,
    thickness: f32,
    color: Srgba,
}

struct PointCloud {
    instances: Vec<PointInstance>,
    object: Option<Gm<Mesh, ColorMaterial>>,
}

struct LineCloud {
    instances: Vec<LineInstance>,
    object: Option<Gm<Mesh, ColorMaterial>>,
}

impl PointCloud {
    fn new() -> Self {
        Self {
            instances: Vec::new(),
            object: None,
        }
    }

    fn clear(&mut self) {
        self.instances.clear();
        self.object = None;
    }

    fn push(
        &mut self,
        position: Vec3,
        radius: f32,
        color: Srgba,
    ) {
        self.instances.push(PointInstance {
            position,
            radius,
            color,
        });
    }

    fn upload(&mut self, context: &Context) {

        if self.instances.is_empty() {
            self.object = None;
            return;
        }

        let template = CpuMesh::sphere(8);

        let template_positions = match &template.positions {
            Positions::F32(p) => p,
            _ => panic!("unexpected sphere format"),
        };

        let template_indices = match &template.indices {
            Indices::U8(i) => i.iter().map(|x| *x as u32).collect::<Vec<_>>(),
            Indices::U16(i) => i.iter().map(|x| *x as u32).collect::<Vec<_>>(),
            Indices::U32(i) => i.clone(),
            _ => panic!("unexpected sphere indices"),
        };

        let mut positions = Vec::<Vec3>::new();
        let mut indices = Vec::<u32>::new();
        let mut colors = Vec::<Srgba>::new();

        for point in &self.instances {

            let base = positions.len() as u32;

            for p in template_positions {

                positions.push(
                    *p * point.radius + point.position
                );

                colors.push(point.color);
            }

            for idx in &template_indices {
                indices.push(base + *idx);
            }
        }

        let cpu_mesh = CpuMesh {
            positions: Positions::F32(positions),
            indices: Indices::U32(indices),
            colors: Some(colors),
            normals: None,
            tangents: None,
            uvs: None,
        };

        let material = ColorMaterial {
            color: Srgba::WHITE,
            render_states: render_states(),
            ..Default::default()
        };

        self.object = Some(
            Gm::new(
                Mesh::new(context, &cpu_mesh),
                material,
            )
        );
    }
}

impl LineCloud {

    fn new() -> Self {
        Self {
            instances: Vec::new(),
            object: None,
        }
    }

    fn clear(&mut self) {
        self.instances.clear();
        self.object = None;
    }

    fn push(
        &mut self,
        a: Vec3,
        b: Vec3,
        thickness: f32,
        color: Srgba,
    ) {
        self.instances.push(LineInstance {
            a,
            b,
            thickness,
            color,
        });
    }

    fn upload(
        &mut self,
        context: &Context,
    ) {

        if self.instances.is_empty() {
            self.object = None;
            return;
        }

        // Continue here...
        //
        // The remainder is essentially the same algorithm:
        //
        // 1. CpuMesh::cylinder(8)
        // 2. Extract positions + indices
        // 3. For every line:
        //      compute rotation exactly as your old stretched_cylinder()
        //      transform every template vertex
        //      append indices with an offset
        //      append colors
        // 4. Construct one CpuMesh
        // 5. Mesh::new(context, ...)
        // 6. Store in self.object
    }
}

fn render_states() -> RenderStates {
    RenderStates {
        depth_test: DepthTest::Less,
        cull: Cull::None,
        ..Default::default()
    }

use anyhow::Result;
use three_d::*;

use crate::camera::OrbitCamera;
use crate::debug::DebugRenderer;
use crate::model::Model;

pub struct Viewer {
    pub model: Model,

    window: Window,
    context: Context,

    cam: OrbitCamera,
    debug: DebugRenderer,
    gui: GUI,

    ambient: AmbientLight,
    directional: DirectionalLight,

    mesh: Gm<Mesh, PhysicalMaterial>,
}

impl Viewer {
    pub fn new(path: &str) -> Result<Self> {
        let window = Window::new(WindowSettings {
            title: "STL Viewer".to_string(),
            max_size: Some((1920, 1080)),
            ..Default::default()
        })?;

        let context = window.gl();

        let model = Model::load(path)?;

        let viewport = window.viewport();
        let cam = OrbitCamera::new(viewport, model.centre, model.radius);
        let gui = GUI::new(&context);

        let ambient = AmbientLight::new(&context, 0.3, Srgba::WHITE);
        let directional = DirectionalLight::new(
            &context,
            2.0,
            Srgba::WHITE,
            vec3(-1.0, -2.0, -3.0),
        );

        let cpu_mesh = model.cpu_mesh();
        let gpu_mesh = Mesh::new(&context, &cpu_mesh);
        let mut material = PhysicalMaterial::new_opaque(
            &context,
            &CpuMaterial {
                albedo: Srgba::new(180, 185, 210, 255),
                ..Default::default()
            },
        );
        material.render_states.cull = Cull::None;

        let mesh = Gm::new(gpu_mesh, material);

        Ok(Self {
            model,
            window,
            context,
            cam,
            debug: DebugRenderer::new(),
            gui,
            ambient,
            directional,
            mesh,
        })
    }

    pub fn sync_mesh(&mut self) {
        let positions: Vec<Vec3> = self
            .model
            .mesh
            .vertices
            .iter()
            .map(|v| vec3(v[0], v[1], v[2]))
            .collect();

        self.mesh.geometry.set_positions(&positions).unwrap();
    }
    
    pub fn run(mut self) -> Result<()> {
        self.window.render_loop(move |mut frame_input| {
            self.cam.resize(frame_input.viewport);
            self.cam.update(&mut frame_input.events);
            self.cam.tighten_clip_planes(self.model.radius);
            
            let fps = if frame_input.elapsed_time > 0.0 {
                1000.0 / frame_input.elapsed_time
            } else {
                0.0
            };

            self.gui.update(
                &mut frame_input.events,
                frame_input.accumulated_time,
                frame_input.viewport,
                frame_input.device_pixel_ratio,
                |gui_context| build_gui(&self.model, &self.cam, gui_context, fps),
            );

            self.debug.clear();
            self.debug.point(
                self.model.centre,
                self.model.radius * 0.02,
                Srgba::new(255, 220, 0, 255),
                true,
            );
            self.debug.edge(
                self.model.centre,
                self.model.centre + vec3(self.model.radius, 0.0, 0.0),
                self.model.radius * 0.01,
                Srgba::new(255, 80, 80, 200),
                false,
            );
            let l = self.model.mesh.vertices.len();
            for i in 0..1000 {
                let i = i * l / 1000;
                self.debug.point(Vector3::from(self.model.mesh.vertices[i].0), self.model.radius*0.005, Srgba::RED, true);
            }

            let BuildResult(occluded, overlay) = self.debug.build(&self.context);

            let _ = frame_input
                .screen()
                .clear(ClearState::color_and_depth(0.09, 0.09, 0.12, 1.0, 1.0))
                .render(
                    &self.cam.camera,
                    std::iter::once(&self.mesh as &dyn Object)
                        .chain(occluded.iter().map(|o| o as &dyn Object)),
                    &[&self.ambient, &self.directional],
                )
                .clear(ClearState::depth(1.0))
                .render(
                    &self.cam.camera,
                    overlay.iter().map(|o| o as &dyn Object),
                    &[&self.ambient, &self.directional],
                )
                // egui renders last, on top of the 3D scene and your
                // existing debug overlay — it's a separate render pass
                // that draws its own UI texture/mesh, no depth clear needed.
                .write(|| self.gui.render());

            FrameOutput::default()
        });
        
        Ok(())
    }
}

fn build_gui(model: &Model, cam: &OrbitCamera, gui_context: &egui::Context, fps: f64) {
    egui::Window::new("Debug")
        .default_pos(egui::pos2(10.0, 60.0))
        .resizable(false)
        .show(gui_context, |ui| {
            ui.label(format!("FPS: {fps:.1}"));
            ui.label(format!("Faces: {}", model.mesh.faces.len()));
            ui.label(format!("Vertices: {}", model.mesh.vertices.len()));
            ui.label(format!(
                "Camera distance: {:.3}",
                cam.camera.position().distance(model.centre)
            ));
            ui.separator();
            ui.label("Drag: orbit   Scroll: zoom");
        });

    let painter = gui_context.debug_painter();

    painter.rect_stroke(
        egui::Rect::from_min_size(egui::pos2(50.0, 400.0), egui::vec2(220.0, 100.0)),
        4.0,
        egui::Stroke::new(2.0, egui::Color32::from_rgb(255, 160, 40)),
        egui::StrokeKind::Middle,
    );

    painter.rect_filled(
        egui::Rect::from_min_size(egui::pos2(320.0, 400.0), egui::vec2(120.0, 60.0)),
        4.0,
        egui::Color32::from_rgba_unmultiplied(80, 180, 255, 160),
    );

    painter.line_segment(
        [egui::pos2(50.0, 540.0), egui::pos2(440.0, 600.0)],
        egui::Stroke::new(3.0, egui::Color32::from_rgb(255, 80, 80)),
    );

    painter.text(
        egui::pos2(50.0, 380.0),
        egui::Align2::LEFT_BOTTOM,
        "raw painter overlay",
        egui::FontId::monospace(14.0),
        egui::Color32::WHITE,
    );
}```

src/model.rs:

use three_d::InnerSpace;
use anyhow::{Context, Result};
use stl_io::{IndexedMesh, Vector};
use std::{
    fs::File,
    io::BufReader,
};

use three_d::{
    CpuMesh,
    Indices,
    Positions,
    Vec3,
};

/// Convert STL coordinates (Z-up) into viewer coordinates (Y-up).
#[inline]
fn zup_to_yup(v: Vector<f32>) -> Vector<f32> {
    Vector::new([
        v[0],
        v[2],
        -v[1],
    ])
}

pub struct Model {
    /// This is the mesh all of your geometry algorithms should use.
    pub mesh: IndexedMesh,

    pub centre: Vec3,
    pub radius: f32,
}

impl Model {
    pub fn load(path: &str) -> Result<Self> {
        let file = File::open(path)
            .with_context(|| format!("Couldn't open STL file '{}'", path))?;

        let mut mesh = stl_io::read_stl(&mut BufReader::new(file))
            .context("Couldn't parse STL")?;

        // Convert to Y-up once so the rest of the program
        // never has to care about coordinate systems.
        for v in &mut mesh.vertices {
            *v = zup_to_yup(*v);
        }

        for face in &mut mesh.faces {
            face.normal = zup_to_yup(face.normal);
        }

        let (centre, radius) = bounding_sphere(&mesh);

        println!(
            "Loaded {} faces, {} vertices",
            mesh.faces.len(),
            mesh.vertices.len()
        );

        Ok(Self {
            mesh,
            centre,
            radius,
        })
    }

    /// Convert our IndexedMesh into a three-d CpuMesh.
    ///
    /// This is only used by the renderer.
    /// Your geometry algorithms should ignore it completely.
    pub fn cpu_mesh(&self) -> CpuMesh {
        let positions: Vec<Vec3> = self.mesh
            .vertices
            .iter()
            .map(|v| Vec3::new(v[0], v[1], v[2]))
            .collect();

        let mut indices = Vec::<u32>::with_capacity(
            self.mesh.faces.len() * 3
        );

        for face in &self.mesh.faces {
            indices.push(face.vertices[0] as u32);
            indices.push(face.vertices[1] as u32);
            indices.push(face.vertices[2] as u32);
        }

        let mut mesh = CpuMesh {
            positions: Positions::F32(positions),
            indices: Indices::U32(indices),

            normals: None,
            tangents: None,
            uvs: None,
            colors: None,
        };

        // Compute smooth normals.
        // Later we can replace this with flat shading if desired.
        mesh.compute_normals();

        mesh
    }
}

/// Recompute the bounding sphere after editing vertices.
pub fn bounding_sphere(mesh: &IndexedMesh) -> (Vec3, f32) {
    let mut lo = Vec3::new(
        f32::INFINITY,
        f32::INFINITY,
        f32::INFINITY,
    );

    let mut hi = Vec3::new(
        f32::NEG_INFINITY,
        f32::NEG_INFINITY,
        f32::NEG_INFINITY,
    );

    for v in &mesh.vertices {
        let p = Vec3::new(
            v[0],
            v[1],
            v[2],
        );

        lo.x = lo.x.min(p.x);
        lo.y = lo.y.min(p.y);
        lo.z = lo.z.min(p.z);

        hi.x = hi.x.max(p.x);
        hi.y = hi.y.max(p.y);
        hi.z = hi.z.max(p.z);
    }

    let centre = (lo + hi) * 0.5;
    let radius = (hi - lo).magnitude() * 0.5;

    (centre, radius)
}```
